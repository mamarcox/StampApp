import { 
  StyleSheet, View, Text, SafeAreaView, ScrollView, 
  TouchableOpacity, Dimensions, StatusBar, Image, TextInput, Alert
} from 'react-native';

  Home, User, ScanLine, ArrowLeft, Award, QrCode, Sparkles, CheckCircle, 
  Ticket, Edit3, Plus, Trash2, Save, LayoutDashboard, ChevronRight, PlusCircle

const COLORS = { blue: '#A2D5EA', yellow: '#F4CC60', brown: '#7C6154', dark: '#381D0E', bg: '#FDFBF7' };

// 资源映射
const STAMP_IMAGES: { [key: string]: any } = {
  '01': require('./assets/images/01.png'),
  '02': require('./assets/images/02.png'),
  '03': require('./assets/images/03.png'),
  '04': require('./assets/images/04.png'),
  '05': require('./assets/images/05.png'),
  '06': require('./assets/images/06.png'),
  '07': require('./assets/images/07.png'),
};

const INITIAL_EVENTS = [
  {
    id: 'ev_001',
    title: "iW Workshop",
    subtitle: "2026 创新学术周",
    coverColor: COLORS.blue,
    stamps: [
      { id: 1, name: "设计思维", collected: false, imgKey: '01' },
      { id: 2, name: "AI 绘画", collected: false, imgKey: '02' },
    ]
  }
];

export default function App() {
  const [role, setRole] = useState<'user' | 'admin'>('user');
  const [currentTab, setCurrentTab] = useState('home');
  const [activeEventId, setActiveEventId] = useState<string | null>(null);
  const [stampPadData, setStampPadData] = useState<any>(null);
  const [showReward, setShowReward] = useState(false); 
  const [lastStampedImg, setLastStampedImg] = useState<any>(null); 
  const [userEvents, setUserEvents] = useState(INITIAL_EVENTS);

  // --- 商家核心逻辑：增删改 ---
  const handleUpdateEvent = (eventId: string, newData: any) => {
    setUserEvents(prev => prev.map(ev => ev.id === eventId ? { ...ev, ...newData } : ev));
  };

  const handleAddEvent = () => {
    const newEvent = {
      id: `ev_${Date.now()}`,
      title: "新活动名称",
      subtitle: "活动简短描述",
      coverColor: [COLORS.blue, COLORS.yellow, '#CDE7B0'][Math.floor(Math.random()*3)],
      stamps: [{ id: Date.now(), name: "首个印章点", collected: false, imgKey: '01' }]
    };
    setUserEvents(prev => [...prev, newEvent]);
  };

  const handleDeleteEvent = (eventId: string) => {
    Alert.alert("删除活动", "确定要永久删除这个活动吗？", [
      { text: "取消", style: "cancel" },
      { text: "确定删除", style: "destructive", onPress: () => {
        setUserEvents(prev => prev.filter(ev => ev.id !== eventId));
        if (activeEventId === eventId) setActiveEventId(null);
      }}
    ]);
  };

  // --- 用户端盖章成功逻辑 ---
  const handleStampSuccess = () => {
    if (!stampPadData || !activeEventId) return;
    setUserEvents(prev => prev.map(event => {
      if (event.id === activeEventId) {
        return { ...event, stamps: event.stamps.map(s => s.id === stampPadData.id ? { ...s, collected: true } : s) };
      }
      return event;
    }));
    setLastStampedImg(STAMP_IMAGES[stampPadData.imgKey]); 
    setStampPadData(null);
  };

  const activeEvent = userEvents.find(e => e.id === activeEventId);

  return (
    <View style={styles.container}>
      <SafeAreaView style={{flex: 1}}>
        <StatusBar barStyle="dark-content" />
        
        {/* 角色切换 */}
        <View style={styles.topBar}>
          <View style={styles.roleSwitcher}>
            <TouchableOpacity onPress={() => setRole('user')} style={[styles.roleBtn, role==='user' && styles.roleBtnActive]}>
              <Text style={[styles.roleBtnText, role==='user' && styles.roleBtnTextActive]}>用户体验端</Text>
            </TouchableOpacity>
            <TouchableOpacity onPress={() => setRole('admin')} style={[styles.roleBtn, role==='admin' && styles.roleBtnActive]}>
              <Text style={[styles.roleBtnText, role==='admin' && styles.roleBtnTextActive]}>商家管理端</Text>
            </TouchableOpacity>
          </View>
        </View>

        {role === 'user' ? (
          /* ================= 用户端：全功能保留 ================= */
          <View style={{flex: 1}}>
            {lastStampedImg ? (
              <View style={styles.successOverlay}>
                <View style={styles.successPaper}>
                  <Sparkles color={COLORS.yellow} size={40} />
                  <Text style={styles.successTitle}>点亮成功！</Text>
                  <View style={styles.successImageWrapper}><Image source={lastStampedImg} style={styles.largeStampImg} resizeMode="contain" /></View>
                  <TouchableOpacity style={styles.successConfirmBtn} onPress={() => setLastStampedImg(null)}><Text style={styles.successConfirmText}>收进印章本</Text></TouchableOpacity>
                </View>
              </View>
            ) : stampPadData ? (
              <StampPadSim name={stampPadData.name} imgKey={stampPadData.imgKey} onBack={() => setStampPadData(null)} onSuccess={handleStampSuccess} />
            ) : showReward ? (
              <RewardScreen onBack={() => setShowReward(false)} />
            ) : activeEvent ? (
              <View style={styles.fullScreen}>
                <ScrollView bounces={false}>
                  <View style={[styles.detailHeader, { backgroundColor: activeEvent.coverColor }]}>
                    <TouchableOpacity onPress={() => setActiveEventId(null)} style={styles.backIcon}><ArrowLeft color={COLORS.dark}/></TouchableOpacity>
                    <Text style={styles.detailTitle}>{activeEvent.title}</Text>
                    <View style={styles.headerStatsRow}>
                      <View style={styles.progressBadge}><Text style={styles.progressText}>进度: {activeEvent.stamps.filter(s=>s.collected).length}/{activeEvent.stamps.length}</Text></View>
                      {activeEvent.stamps.every(s=>s.collected) && (
                        <TouchableOpacity style={styles.redeemBtn} onPress={() => setShowReward(true)}>
                          <Ticket color={COLORS.yellow} size={16} /><Text style={styles.redeemBtnText}>核销奖励</Text>
                        </TouchableOpacity>
                      )}
                    </View>
                  </View>
                  <View style={styles.detailTransition} />
                  <View style={styles.stampGrid}>
                    {activeEvent.stamps.map((stamp, i) => (
                      <TouchableOpacity key={stamp.id} disabled={stamp.collected} onPress={() => setStampPadData(stamp)} style={[styles.stampCard, stamp.collected ? styles.stampCollected : styles.stampUncollected]}>
                        <View style={styles.stampCardHeader}>
                          <Text style={[styles.stampNum, {color: stamp.collected ? COLORS.dark : '#CCC'}]}>{String(i+1).padStart(2,'0')}</Text>
                          {stamp.collected && <CheckCircle size={20} color={COLORS.dark} />}
                        </View>
                        <View style={styles.stampImagePlaceholder}>
                          {stamp.collected ? <Image source={STAMP_IMAGES[stamp.imgKey]} style={styles.gridStampImg} /> : <Text style={styles.stampLockIcon}>?</Text>}
                        </View>
                        <Text style={styles.stampNameText}>{stamp.name}</Text>
                      </TouchableOpacity>
                    ))}
                  </View>
                </ScrollView>
              </View>
            ) : (
              <View style={{flex: 1}}>
                <ScrollView style={styles.content}>
                  <Text style={styles.welcomeText}>Welcome back,</Text>
                  <Text style={styles.explorerText}>Explorer</Text>
                  <Text style={styles.sectionTitle}>✨ 正在进行的集章</Text>
                  {userEvents.map(event => (
                    <TouchableOpacity key={event.id} onPress={() => setActiveEventId(event.id)} style={[styles.heroCard, { backgroundColor: event.coverColor }]}>
                      <Text style={styles.heroTitle}>{event.title}</Text>
                      <View style={styles.heroFooter}>
                        <View style={styles.miniStamps}>
                          {event.stamps.map((s, idx) => (
                            <View key={idx} style={[styles.miniCircle, {backgroundColor: 'white', overflow:'hidden', borderWidth: 2, borderColor: COLORS.dark, marginRight: -8}]}>
                              {s.collected && <Image source={STAMP_IMAGES[s.imgKey]} style={{width:'100%', height:'100%'}} />}
                            </View>
                          ))}
                        </View>
                        <View style={styles.goBtn}><Text style={styles.goText}>查看详情</Text></View>
                      </View>
                    </TouchableOpacity>
                  ))}
                </ScrollView>
                <View style={styles.bottomNav}>
                  <TouchableOpacity onPress={() => setCurrentTab('home')}><Home color={currentTab==='home'?COLORS.dark:'#CCC'} /></TouchableOpacity>
                  <TouchableOpacity style={styles.mainFab}><ScanLine color={COLORS.dark} size={32} /></TouchableOpacity>
                  <TouchableOpacity onPress={() => setCurrentTab('profile')}><User color={currentTab==='profile'?COLORS.dark:'#CCC'} /></TouchableOpacity>
                </View>
              </View>
            )}
          </View>
        ) : (
          /* ================= 商家端：新增活动逻辑 ================= */
          <AdminPanel 
            events={userEvents} 
            onUpdate={handleUpdateEvent} 
            onAdd={handleAddEvent} 
            onDelete={handleDeleteEvent} 
          />
        )}
      </SafeAreaView>
    </View>
  );
}

function AdminPanel({ events, onUpdate, onAdd, onDelete }: any) {
  const [editingId, setEditingId] = useState<string | null>(null);

  return (
    <ScrollView style={styles.adminContainer}>
      <View style={styles.adminHeaderRow}>
        <LayoutDashboard color={COLORS.dark} size={28} />
        <Text style={styles.adminHeaderText}>控制台</Text>
      </View>
      
      {/* 新增活动按钮 */}
      <TouchableOpacity style={styles.createEventBtn} onPress={onAdd}>
        <PlusCircle color="white" size={24} />
        <Text style={styles.createEventText}>创建新集章活动</Text>
      </TouchableOpacity>

      {events.map((event: any) => (
        <View key={event.id} style={styles.adminEventCard}>
          {editingId === event.id ? (
            <View>
              <View style={styles.adminCardHeader}>
                <Edit3 size={18} color={COLORS.dark} />
                <Text style={styles.adminCardTitle}> 编辑活动</Text>
                <TouchableOpacity style={{marginLeft: 'auto'}} onPress={() => onDelete(event.id)}>
                  <Trash2 size={20} color="#FF6B6B" />
                </TouchableOpacity>
              </View>
              
              <Text style={styles.inputLabel}>活动名称</Text>
              <TextInput style={styles.adminInput} value={event.title} onChangeText={(t) => onUpdate(event.id, {title: t})} />
              
              <Text style={styles.inputLabel}>印章列表</Text>
              {event.stamps.map((stamp: any, index: number) => (
                <View key={stamp.id} style={styles.adminStampRow}>
                  <TextInput style={[styles.adminInput, {flex: 1, marginBottom: 0}]} value={stamp.name} onChangeText={(t) => {
                    const newStamps = [...event.stamps];
                    newStamps[index].name = t;
                    onUpdate(event.id, {stamps: newStamps});
                  }} />
                  <TouchableOpacity style={styles.imgPicker} onPress={() => {
                    const nextImg = (parseInt(stamp.imgKey) % 7 + 1).toString().padStart(2, '0');
                    const newStamps = [...event.stamps];
                    newStamps[index].imgKey = nextImg;
                    onUpdate(event.id, {stamps: newStamps});
                  }}><Image source={STAMP_IMAGES[stamp.imgKey]} style={{width: 24, height: 24}} /></TouchableOpacity>
                  <TouchableOpacity onPress={() => {
                    const newStamps = event.stamps.filter((s: any) => s.id !== stamp.id);
                    onUpdate(event.id, {stamps: newStamps});
                  }}><Trash2 size={18} color="#FF6B6B" /></TouchableOpacity>
                </View>
              ))}
              <TouchableOpacity style={styles.addStampBtn} onPress={() => {
                const newStamp = { id: Date.now(), name: "新印章", collected: false, imgKey: '01' };
                onUpdate(event.id, {stamps: [...event.stamps, newStamp]});
              }}><Plus size={16} color={COLORS.dark} /><Text style={styles.addStampBtnText}>添加印章位</Text></TouchableOpacity>

              <TouchableOpacity style={styles.adminSaveBtn} onPress={() => setEditingId(null)}>
                <Save size={18} color="white" /><Text style={styles.adminSaveBtnText}>保存修改</Text>
              </TouchableOpacity>
            </View>
          ) : (
            <TouchableOpacity style={styles.adminSummary} onPress={() => setEditingId(event.id)}>
              <View style={{flex: 1}}><Text style={styles.adminSummaryTitle}>{event.title}</Text>
              <Text style={styles.adminSummarySub}>{event.stamps.length} 个集章位</Text></View>
              <ChevronRight size={24} color={COLORS.dark} />
            </TouchableOpacity>
          )}
        </View>
      ))}
    </ScrollView>
  );
}

// --- 物理盖章识别组件 ---
function StampPadSim({ name, imgKey, onBack, onSuccess }: any) {
  const [progress, setProgress] = useState(0);
  const timer = useRef<any>(null);
  const handleTouch = (evt: any) => {
    if (evt.nativeEvent.touches.length === 4) {
      if (!timer.current) {
        timer.current = setInterval(() => {
          setProgress(p => {
            if (p >= 100) { clearInterval(timer.current); timer.current = null; onSuccess(); return 100; }
            return p + 4;
          });
        }, 40);
      }
    } else {
      if (timer.current) { clearInterval(timer.current); timer.current = null; setProgress(0); }
    }
  };
  return (
    <View style={styles.padContainer} onStartShouldSetResponder={() => true} onResponderMove={handleTouch}>
      <TouchableOpacity onPress={onBack} style={styles.padBack}><ArrowLeft color="white" /></TouchableOpacity>
      <Text style={styles.padTitle}>HARDWARE MODE</Text>
      <Text style={styles.padName}>{name}</Text>
      <View style={styles.circleWrapper}>
        <Svg style={styles.svgLayer}><Circle cx="110" cy="110" r="100" stroke="rgba(255,255,255,0.05)" strokeWidth="12" fill="none" />
        <Circle cx="110" cy="110" r="100" stroke={COLORS.yellow} strokeWidth="12" fill="none" strokeDasharray="628" strokeDashoffset={628 - (628 * progress) / 100} /></Svg>
        <View style={styles.innerCircle}><Image source={STAMP_IMAGES[imgKey]} style={{width: 100, height: 100, opacity: 0.2 + progress/100}} resizeMode="contain" /></View>
      </View>
    </View>
  );
}

      <TouchableOpacity onPress={onBack} style={styles.rewardBackBtn}><Text style={styles.rewardBackText}>返回印章本</Text></TouchableOpacity>
    </View>
  );
}
